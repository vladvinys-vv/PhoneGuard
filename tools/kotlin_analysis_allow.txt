# Ложные срабатывания проверки «suspend-вызов вне suspend-контекста»
#
# tools/kotlin_analysis.py собирает имена suspend-функций по всему проекту.
# Если в другом классе есть не-suspend функция с тем же именем, проверка
# срабатывает ложно. Каждая строка ниже проверена вручную: в правой части —
# имя символа, в левой — файл; формат `путь : имя`.
#
# Правило: добавлять сюда только после того, как убедились, что вызывается
# именно не-suspend перегрузка. Новые записи без пояснения не принимаются.

# --- UI: ViewModel-методы с тем же именем, что у suspend-методов сервисов ----
# stopAlarm: suspend есть у SecurityService, а AntiTheftViewModel.stopAlarm() — обычная.
app/src/main/java/com/phoneguard/ui/screens/antitheft/AntiTheftScreen.kt: stopAlarm
# upsertRule: suspend в FirewallRepository, FirewallViewModel.upsertRule() — обычная.
app/src/main/java/com/phoneguard/ui/screens/firewall/FirewallScreen.kt: upsertRule
# persist: suspend в SettingsRepository, а это локальная функция composable-скоупа.
app/src/main/java/com/phoneguard/ui/screens/firewall/FirewallScreen.kt: persist
# setDarkTheme / setLanguage / setThreatIntelEnabled: suspend в PreferencesManager,
# у SettingsViewModel все три запускают viewModelScope.launch и сами не suspend.
app/src/main/java/com/phoneguard/ui/screens/settings/SettingsScreen.kt: setDarkTheme
app/src/main/java/com/phoneguard/ui/screens/settings/SettingsScreen.kt: setLanguage
app/src/main/java/com/phoneguard/ui/screens/settings/SettingsScreen.kt: setThreatIntelEnabled
# exportLogs: suspend в SettingsRepository, SettingsViewModel.exportLogs() — запуск корутины.
app/src/main/java/com/phoneguard/ui/screens/settings/SettingsScreen.kt: exportLogs
# deleteItem: suspend в VaultRepository, VaultViewModel.deleteItem() — запуск корутины.
app/src/main/java/com/phoneguard/ui/screens/vault/VaultScreen.kt: deleteItem

# --- util ------------------------------------------------------------------
